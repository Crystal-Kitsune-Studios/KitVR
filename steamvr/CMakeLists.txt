cmake_minimum_required(VERSION 3.22)
project(kitvr_steamvr_driver LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(OPENVR_DRIVER_HEADER "${CMAKE_CURRENT_BINARY_DIR}/openvr/openvr_driver.h")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/openvr")
file(DOWNLOAD
    "https://raw.githubusercontent.com/ValveSoftware/openvr/v2.15.6/headers/openvr_driver.h"
    "${OPENVR_DRIVER_HEADER}"
    EXPECTED_HASH SHA256=1036efe998d63e82d1d3db2b32a2f58df4a8eeaf5280f50aaf28220ff60a40ab
    TLS_VERIFY ON
    STATUS OPENVR_DOWNLOAD_STATUS)
list(GET OPENVR_DOWNLOAD_STATUS 0 OPENVR_DOWNLOAD_CODE)
if(NOT OPENVR_DOWNLOAD_CODE EQUAL 0)
    list(GET OPENVR_DOWNLOAD_STATUS 1 OPENVR_DOWNLOAD_MESSAGE)
    message(FATAL_ERROR "Could not fetch OpenVR driver header: ${OPENVR_DOWNLOAD_MESSAGE}")
endif()

add_library(driver_kitvr SHARED src/driver.cpp)
target_include_directories(driver_kitvr PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/openvr")
target_compile_definitions(driver_kitvr PRIVATE OPENVR_BUILD_STATIC)
include(CTest)
if(BUILD_TESTING)
    if(NOT WIN32)
        add_executable(kitvr_view_config_test tests/view_config_test.cpp)
        target_include_directories(kitvr_view_config_test PRIVATE src)
        add_test(NAME kitvr_view_config_test COMMAND kitvr_view_config_test)
        add_executable(kitvr_video_frame_test tests/video_frame_test.cpp)
        add_test(NAME kitvr_video_frame_test COMMAND kitvr_video_frame_test)
        add_executable(kitvr_hand_state_test tests/hand_state_test.cpp)
        target_include_directories(kitvr_hand_state_test PRIVATE src)
        add_test(NAME kitvr_hand_state_test COMMAND kitvr_hand_state_test)
        add_executable(kitvr_controller_pose_test tests/controller_pose_test.cpp)
        target_include_directories(kitvr_controller_pose_test PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/openvr")
        target_compile_definitions(kitvr_controller_pose_test PRIVATE OPENVR_BUILD_STATIC)
        add_test(NAME kitvr_controller_pose_test COMMAND kitvr_controller_pose_test)
        add_executable(kitvr_hand_input_test ../quest/tests/hand_input_test.cpp)
        target_include_directories(kitvr_hand_input_test PRIVATE ../quest/app/src/main/cpp)
        add_test(NAME kitvr_hand_input_test COMMAND kitvr_hand_input_test)
        add_executable(kitvr_tracking_sequence_test tests/tracking_sequence_test.cpp)
        target_include_directories(kitvr_tracking_sequence_test PRIVATE src)
        add_test(NAME kitvr_tracking_sequence_test COMMAND kitvr_tracking_sequence_test)
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            find_package(Threads REQUIRED)
            add_executable(kitvr_control_send_test ../quest/tests/control_send_test.cpp)
            target_include_directories(kitvr_control_send_test PRIVATE ../quest/app/src/main/cpp)
            add_test(NAME kitvr_control_send_test COMMAND kitvr_control_send_test)
            add_executable(kitvr_session_reconnect_test
                ../quest/tests/session_reconnect_test.cpp ../quest/app/src/main/cpp/session.cpp)
            target_include_directories(kitvr_session_reconnect_test PRIVATE
                ../quest/tests/stubs ../quest/app/src/main/cpp)
            target_link_libraries(kitvr_session_reconnect_test PRIVATE Threads::Threads)
            target_link_options(kitvr_session_reconnect_test PRIVATE "-Wl,--wrap=send")
            add_test(NAME kitvr_session_reconnect_test COMMAND kitvr_session_reconnect_test)
            set_tests_properties(kitvr_session_reconnect_test PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 20)
            add_executable(kitvr_decoder_output_test
                ../quest/tests/decoder_output_test.cpp ../quest/app/src/main/cpp/decoder.cpp)
            target_include_directories(kitvr_decoder_output_test PRIVATE
                ../quest/tests/stubs ../quest/app/src/main/cpp)
            target_link_libraries(kitvr_decoder_output_test PRIVATE Threads::Threads)
            add_test(NAME kitvr_decoder_output_test COMMAND kitvr_decoder_output_test)
            set_tests_properties(kitvr_decoder_output_test PROPERTIES TIMEOUT 10)
        endif()
    endif()
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME kitvr_input_bindings_test
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/input_bindings_test.py")
    endif()
endif()
if(WIN32)
    target_link_libraries(driver_kitvr PRIVATE ws2_32)
    set_target_properties(driver_kitvr PROPERTIES PREFIX "")
    install(TARGETS driver_kitvr RUNTIME DESTINATION kitvr/bin/win64)
else()
    set_target_properties(driver_kitvr PROPERTIES PREFIX "")
    install(TARGETS driver_kitvr LIBRARY DESTINATION kitvr/bin/linux64)
endif()

install(FILES driver/driver.vrdrivermanifest DESTINATION kitvr)
install(DIRECTORY driver/resources/ DESTINATION kitvr/resources)
install(FILES README.md DESTINATION kitvr)
